% Figure: CMDGFM rows (Chapter 10). Cells, solid, liquid polygons, PLIC
% segments, the classification of the mixed faces and one CMDGFM stencil are
% read from the code: generated/projection.tex (projection.jl,
% projection_data). Labels hand-written.
\begin{figure}
  \centering
  \def\fxpnone{%
    \def\fxpcell##1##2##3##4##5##6##7{}\def\fxpsolid##1##2{}\def\fxpwall##1##2##3##4##5{}%
    \def\fxpliquid##1##2{}\def\fxpplic##1##2##3##4##5{}%
    \def\fxpclass##1##2##3##4##5{}\def\fxpstencil##1##2##3##4##5{}\def\fxpnormal##1##2##3##4{}}%
  % A face segment of axis #3 and length #4 at (#1,#2), drawn with style #5.
  \def\fxpsegment#1#2#3#4#5{%
    \ifnum#3=1 \draw[#5] (#1,#2-#4/2) -- (#1,#2+#4/2);
    \else \draw[#5] (#1-#4/2,#2) -- (#1+#4/2,#2);\fi}%
  % Background of panel #1: levels, liquid, solid, cell lines, walls, PLIC.
  \def\fxpbackground#1{%
    \fxpnone
    \def\fxpcell##1##2##3##4##5##6##7{\ifnum##1=#1
      \ifnum##7=2 \fill[tr level ##6] (##2,##3) rectangle (##4,##5);
        \fill[tr solid] (##2,##3) rectangle (##4,##5);
      \else \fill[tr level ##6] (##2,##3) rectangle (##4,##5);\fi\fi}%
    \input{figures/generated/projection.tex}%
    \fxpnone
    \def\fxpliquid##1##2{\ifnum##1=#1 \fill[tr liquid] ##2;\fi}%
    \def\fxpsolid##1##2{\ifnum##1=#1 \fill[tr solid] ##2;\fi}%
    \input{figures/generated/projection.tex}%
    \fxpnone
    \def\fxpcell##1##2##3##4##5##6##7{\ifnum##1=#1 \draw[tr cell] (##2,##3) rectangle (##4,##5);\fi}%
    \def\fxpwall##1##2##3##4##5{\ifnum##1=#1 \draw[tr solid line] (##2,##3) -- (##4,##5);\fi}%
    \def\fxpplic##1##2##3##4##5{\ifnum##1=#1 \draw[tr plic] (##2,##3) -- (##4,##5);\fi}%
    \input{figures/generated/projection.tex}%
    \fxpnone}%
  \begin{subfigure}[b]{0.7\textwidth}
    \centering
    \begin{tikzpicture}[x=3.6cm,y=3.6cm]
      \begin{scope}
        \clip (3.25,3.0) rectangle (5.0,4.25);
        \fxpbackground{1}
        % The stencil of one row: f red, transverse faces purple, the
        % inference faces grey (gas) and blue (liquid).
        \def\fxpstencil#1#2#3#4#5{%
          \ifcase#5 \fxpsegment{#1}{#2}{#3}{#4}{tr face}%
          \or \fxpsegment{#1}{#2}{#3}{#4}{line width=1.8pt,trpurple}%
          \or \fxpsegment{#1}{#2}{#3}{#4}{line width=1.8pt,trgrey}%
          \or \fxpsegment{#1}{#2}{#3}{#4}{line width=1.8pt,trblue}\fi}%
        \input{figures/generated/projection.tex}%
        \fxpnone
        % Rows of the mixed faces: beta = 1 (disc), beta < 1 (triangle),
        % unresolved (open circle).
        \def\fxpclass#1#2#3#4#5{%
          \ifcase#5 \node[circle,fill=black,inner sep=0pt,minimum size=3.4pt] at (#1,#2) {};
          \or \node[regular polygon,regular polygon sides=3,fill=black,inner sep=0pt,minimum size=5.2pt] at (#1,#2) {};
          \or \node[circle,draw=black,fill=white,line width=0.6pt,inner sep=0pt,minimum size=3.8pt] at (#1,#2) {};\fi}%
        \input{figures/generated/projection.tex}%
        \fxpnone
        \def\fxpnormal#1#2#3#4{\draw[tr vector] (#1,#2) -- ++(0.3*#3,0.3*#4);}%
        \input{figures/generated/projection.tex}%
        \fxpnone
      \end{scope}
      \draw[black,line width=0.6pt] (3.25,3.0) rectangle (5.0,4.25);
      \draw[black!70,dash pattern=on 2pt off 1.5pt,line width=0.5pt] (4,2.97) -- (4,4.28);
      \node[tr label,anchor=east,fill=white,fill opacity=0.8,text opacity=1] at (4.48,3.875) {$f$};
      \node[tr label,anchor=east,fill=white,fill opacity=0.8,text opacity=1] at (4.48,4.125) {$\inferface{\gas}$};
      \node[tr label,anchor=east,fill=white,fill opacity=0.8,text opacity=1] at (4.48,3.625) {$\inferface{\liq}$};
      \node[tr label,anchor=west,trred,fill=white,fill opacity=0.8,text opacity=1] at (4.56,4.12) {$\normalstag_f$};
      \node[tr label,anchor=west,trpurple,fill=white,fill opacity=0.8,text opacity=1] at (4.76,3.94) {$\transfaces{f}$};
      \node[tr small,anchor=south west,fill=white,fill opacity=0.85,text opacity=1] at (3.27,3.02)
        {$\jumpbeta_f=\chkFigBeta,\ \normratio{f}=\chkFigRatio$};
    \end{tikzpicture}
    \caption{CMDGFM rows}
    \label{fig:projection:stencils:cmdgfm}
  \end{subfigure}
  \caption{Constraint rows of the CMDGFM projection on the composite
    cut fixture of \cref{tab:projection:counts} (a drop on the sloped solid;
    level transition dashed; liquid shaded, solid hatched), near the top of
    the drop. Rows of the mixed faces: disc
    $\jumpbeta_f=1$, triangle $\jumpbeta_f<1$ (with inference faces), open
    circle unresolved~\cref{eq:projection:unresolved}. The stencil of one row
    with $\abs{\normratio{f}}>1$~\cref{eq:projection:cmdgfm-row}: the face $f$
    (red) with its staggered normal $\normalstag_f$, the transverse faces
    $\transfaces{f}$ (purple) and the inference faces $\inferface{\gas}$
    (grey) and $\inferface{\liq}$ (blue).}
  \label{fig:projection:stencils}
\end{figure}
