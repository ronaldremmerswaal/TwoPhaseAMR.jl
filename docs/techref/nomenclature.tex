% Selected and adapted from the thesis nomenclature; AMR entries are marked.
\begin{description}
\item[$\mathcal{B},\tree,\leafset$] Mesh block, block tree, and set of leaf blocks (AMR).
\item[$\level, h_0, h$] Maximum refinement level, base mesh width, and minimum mesh width.
\item[$\hangsegment{f},\alphacf$] Hanging-face segment and cell--face incidence fraction (AMR).
\item[$\cvset{C},\cvset{F}$] Sets of centred control volumes and their faces.
\item[$\funspace{C},\funspace{F}$] Spaces of centred and face functions.
\item[$\omega_f$] Staggered control volume associated with face $f$.
\item[$\volfrac^\liq,\volfrac^\gas$] Liquid and gas volume fractions.
\item[$\chi,\wycflval$] Phase indicator and WY CFL limit.
\item[$\massflux,\dt$] Mass flux and time-step size.
\item[$R,s,x$] Normal-component ratio, inference mass-fraction sum, and normal alignment in CMDGFM.
\item[$\divh,\gradh$] Discrete divergence and gradient.
\item[$u^*,u^{**},u^{***}$] Velocity after transport, diffusion, and forces, respectively.
\item[$\beta,\xi,\lambda$] Jump-interpolation weight, scaled pressure-gradient jump, and KKT multiplier.
\item[$\kappa,\sigma$] Interface curvature and surface-tension coefficient.
\item[$\rho^\liq,\rho^\gas,\+g$] Liquid and gas densities and gravitational acceleration.
\end{description}
